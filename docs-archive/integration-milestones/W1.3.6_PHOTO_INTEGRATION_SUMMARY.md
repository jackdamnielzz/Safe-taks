## Integration Pattern

All steps follow the same consistent pattern established in Step 2:

```typescript
// 1. Import components
import PhotoGallery from "@/components/lmra/PhotoGallery";
import PhotoCapture from "@/components/lmra/PhotoCapture";

// 2. Add props
type Props = {
  lmra?: Partial<LMRA>;
  onChange: (stepPayload: Partial<LMRAStepX>) => void;
  userId?: string;
  userName?: string;
};

// 3. Add state
const [showCamera, setShowCamera] = useState(false);

// 4. Add PhotoGallery (at end of component)
{lmra?.id && userId && (
  <div className="mt-6 pt-6 border-t">
    <PhotoGallery
      lmraId={lmra.id}
      stepNumber={X}
      onAddPhoto={() => setShowCamera(true)}
      maxPhotos={Y}
    />
  </div>
)}

// 5. Add PhotoCapture modal
{showCamera && lmra?.id && userId && (
  <PhotoCapture
    lmraId={lmra.id}
    stepNumber={X}
    userId={userId}
    userName={userName}
    onPhotoAdded={() => setShowCamera(false)}
    onClose={() => setShowCamera(false)}
    maxPhotos={Y}
  />
)}
```

## Storage Structure

### Firebase Storage Paths
```
lmras/{lmraId}/photos/{photoId}.jpg           # Main image
lmras/{lmraId}/photos/thumbnails/{photoId}_thumb.jpg  # Thumbnail
```

### Firestore Collection
```
lmraPhotos/{photoId}
  - lmraId: string
  - stepNumber: number (2, 5, 6, or 7)
  - caption?: string
  - uploadedBy: string (userId)
  - uploadedAt: timestamp
  - metadata: {
      width: number
      height: number
      size: number
      mimeType: string
      latitude?: number
      longitude?: number
    }
  - firebaseUrl: string
  - thumbnailUrl: string
```

### IndexedDB Structure
```
Database: lmra-photos
Store: photos
Indexes:
  - by-lmra: lmraId
  - by-sync-status: syncStatus
  - by-step: [lmraId, stepNumber]
```
